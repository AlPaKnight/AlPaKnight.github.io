// ═══════════════════ GESTION DU THÈME (Dark/Light) ═══════════════════
const themeToggle = document.getElementById('theme-toggle');
const html = document.documentElement;

// Initialiser le thème au chargement
function initTheme() {
  const savedTheme = localStorage.getItem('pdl-theme') || 'light';
  html.setAttribute('data-theme', savedTheme);
  updateThemeIcon(savedTheme);
}

function updateThemeIcon(theme) {
  const iconMoon = themeToggle.querySelector('.icon-moon');
  const iconSun = themeToggle.querySelector('.icon-sun');
  
  if (theme === 'dark') {
    iconMoon.style.display = 'none';
    iconSun.style.display = 'block';
    themeToggle.setAttribute('aria-label', 'Passer en mode clair');
  } else {
    iconMoon.style.display = 'block';
    iconSun.style.display = 'none';
    themeToggle.setAttribute('aria-label', 'Passer en mode sombre');
  }
}

themeToggle.addEventListener('click', () => {
  const currentTheme = html.getAttribute('data-theme');
  const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
  
  html.setAttribute('data-theme', newTheme);
  localStorage.setItem('pdl-theme', newTheme);
  updateThemeIcon(newTheme);
});

initTheme();

// ═══════════════════ GESTION DE LA LANGUE (FR/EN) ═══════════════════
const langToggle = document.getElementById('lang-toggle');
const langButtons = langToggle.querySelectorAll('button');

function initLang() {
  const savedLang = localStorage.getItem('pdl-lang') || 'fr';
  setLanguage(savedLang);
}

function setLanguage(lang) {
  localStorage.setItem('pdl-lang', lang);
  html.setAttribute('lang', lang);
  
  // Mettre à jour les boutons actifs
  langButtons.forEach(btn => {
    btn.classList.toggle('active', btn.dataset.lang === lang);
  });
  
  // Mettre à jour le contenu avec data-fr et data-en
  document.querySelectorAll('[data-fr][data-en]').forEach(el => {
    const text = el.dataset[lang];
    if (text) {
      if (el.tagName === 'INPUT' || el.tagName === 'BUTTON') {
        el.value = text;
        el.textContent = text;
      } else {
        el.textContent = text;
      }
    }
  });
  
  // Mettre à jour les aria-labels
  document.querySelectorAll(`[data-${lang}-label]`).forEach(el => {
    el.setAttribute('aria-label', el.dataset[`${lang}Label`]);
  });
  
  // Mettre à jour les data-open/close pour le bouton modules
  const toggleBtn = document.getElementById('toggle-modules');
  if (toggleBtn) {
    const isOpen = document.getElementById('modules-grid').classList.contains('open');
    const key = isOpen ? 'close' : 'open';
    const text = toggleBtn.dataset[`${lang}${key === 'open' ? 'Open' : 'Close'}`];
    if (text) {
      toggleBtn.querySelector('span').textContent = text;
    }
  }
}

langButtons.forEach(btn => {
  btn.addEventListener('click', () => {
    setLanguage(btn.dataset.lang);
  });
});

initLang();

// ═══════════════════ GESTION DU MENU BURGER MOBILE ═══════════════════
const navBurger = document.querySelector('.nav__burger');
const navLinks = document.querySelector('.nav__links');

navBurger.addEventListener('click', () => {
  const isExpanded = navBurger.getAttribute('aria-expanded') === 'true';
  navBurger.setAttribute('aria-expanded', !isExpanded);
  navLinks.classList.toggle('active');
});

// Fermer le menu quand on clique sur un lien
navLinks.querySelectorAll('a').forEach(link => {
  link.addEventListener('click', () => {
    navBurger.setAttribute('aria-expanded', 'false');
    navLinks.classList.remove('active');
  });
});

// ═══════════════════ MARQUER LE LIEN ACTIF ═══════════════════
function updateActiveLink() {
  const currentPage = window.location.pathname.split('/').pop() || 'index.html';
  document.querySelectorAll('.nav__links a').forEach(link => {
    const href = link.getAttribute('href');
    link.classList.toggle('active', href === currentPage || 
      (currentPage === '' && href === 'index.html'));
  });
}

updateActiveLink();
